class _ProjectsSection extends StatelessWidget {
  const _ProjectsSection({super.key});
  @override
  Widget build(BuildContext context) {
    final w = MediaQuery.of(context).size.width;
    final mobile = w < 600;
    return Padding(
      padding: const EdgeInsets.only(bottom: 96),
      child: Column(children: [
        _ProjectRow(mobile: mobile, reverse: false),
        SizedBox(height: mobile ? 48 : 80),
        _ProjectRow(mobile: mobile, reverse: true),
      ]),
    );
  }
}

class _ProjectRow extends StatelessWidget {
  const _ProjectRow({required this.mobile, required this.reverse});
  final bool mobile, reverse;
  @override
  Widget build(BuildContext context) {
    final words = _ProjectText(alignRight: reverse && !mobile);
    const img = _ProjectImg();
    if (mobile) {
      return Column(crossAxisAlignment: CrossAxisAlignment.start,
          children: [words, const SizedBox(height: 18), img]);
    }
    return Row(crossAxisAlignment: CrossAxisAlignment.start, children: reverse
        ? [const Expanded(child: img), const SizedBox(width: 36), Expanded(child: words)]
        : [Expanded(child: words), const SizedBox(width: 36), const Expanded(child: img)]);
  }
}

class _ProjectText extends StatelessWidget {
  const _ProjectText({required this.alignRight});
  final bool alignRight;
  @override
  Widget build(BuildContext context) {
    final w = MediaQuery.of(context).size.width;
    final mobile = w < 600;
    final cross = alignRight ? CrossAxisAlignment.end : CrossAxisAlignment.start;
    final align = alignRight ? TextAlign.right : TextAlign.left;
    return Column(crossAxisAlignment: cross, children: [
      Text('Featured Project',
          textAlign: align,
          style: TextStyle(
              fontSize: mobile ? 11 : 13,
              color: _violet,
              fontWeight: FontWeight.bold)),
      const SizedBox(height: 6),
      Text('Example Project',
          textAlign: align,
          style: TextStyle(
              fontSize: _fs(w, 22, 28, 34),
              color: const Color(0xffd7d5ff),
              fontWeight: FontWeight.w600)),
      const SizedBox(height: 16),
      Container(
        padding: const EdgeInsets.all(16),
        decoration: BoxDecoration(
            color: const Color(0xee241f42),
            borderRadius: BorderRadius.circular(10)),
        child: Text(
          'A web app for visualizing personalized Spotify data. View your top artists, '
          'top tracks, recently played tracks, and detailed audio information about each track. '
          'Create and save new playlists of recommended tracks based on your existing playlists.',
          textAlign: align,
          style: TextStyle(
              fontSize: mobile ? 12 : 13, height: 1.55, color: const Color(0xffd3d1e9)),
        ),
      ),
      const SizedBox(height: 14),
      const Text('✽  ✽', style: TextStyle(fontSize: 20)),
    ]);
  }
}

class _ProjectImg extends StatefulWidget {
  const _ProjectImg();
  @override
  State<_ProjectImg> createState() => _ProjectImgState();
}

class _ProjectImgState extends State<_ProjectImg> {
  bool _hover = false;
  @override
  Widget build(BuildContext context) {
    final w = MediaQuery.of(context).size.width;
    final h = w < 600 ? 180.0 : 220.0;
    return MouseRegion(
      onEnter: (_) => setState(() => _hover = true),
      onExit: (_) => setState(() => _hover = false),
      child: AnimatedContainer(
        duration: const Duration(milliseconds: 200),
        height: h,
        clipBehavior: Clip.antiAlias,
        decoration: BoxDecoration(
          color: const Color(0xff2b0b3a),
          borderRadius: BorderRadius.circular(10),
          boxShadow: [BoxShadow(
              color: _hover ? const Color(0xbb7127ba) : const Color(0x887127ba),
              blurRadius: _hover ? 56 : 40,
              spreadRadius: 3)],
        ),
        child: Image.asset('assets/figma/project-right.png',
            fit: BoxFit.cover, width: double.infinity),
      ),
    );
  }
}
